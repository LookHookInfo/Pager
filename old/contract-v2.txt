pragma solidity ^0.8.34;

import "@openzeppelin/contracts/token/ERC1155/ERC1155.sol";
import "@openzeppelin/contracts/access/Ownable.sol";
import "@openzeppelin/contracts/token/common/ERC2981.sol";
import "@openzeppelin/contracts/token/ERC20/IERC20.sol";
import "@openzeppelin/contracts/utils/Strings.sol";
import "@openzeppelin/contracts/utils/ReentrancyGuard.sol";

contract PagerMascotKeysV2 is ERC1155, Ownable, ERC2981, ReentrancyGuard {
    using Strings for uint256;

    uint256 public constant MAX_SUPPLY = 10000;
    uint96 public constant ROYALTY_BPS = 2000;
    uint256 public constant CREATION_FEE = 500 * 10**18;
    IERC20 public immutable hashToken;

    string public name = "Pager Mascot Keys";
    string public symbol = "PMK";

    struct MascotKey {
        address creator;
        uint256 price;
        uint32 currentSupply;
        uint32 totalSold;
        bool exists;
    }

    address public teamWallet;
    uint256 public nextTokenId;

    string public collectionMetadata = "ipfs://bafkreibgkoo4zv3mctvyghtzhx5utmperwkfzl4jpwoes2f2ymfefmz3ea";
    string public baseURI;

    mapping(uint256 => MascotKey) public keys;
    uint256[] public allTokenIds;

    mapping(address => uint256[]) private userTokens;
    mapping(address => mapping(uint256 => bool)) private userTokenExists;

    event CreationFeePaid(address indexed creator, uint256 amount);
    event MascotCreated(uint256 indexed tokenId, address indexed creator, uint256 price);
    event MascotMinted(uint256 indexed tokenId, address indexed buyer);
    event MascotDiscarded(uint256 indexed tokenId, address indexed user);
    event MascotDeleted(uint256 indexed tokenId);
    event PriceUpdated(uint256 indexed tokenId, uint256 newPrice);

    constructor(address _hashToken, address _teamWallet, string memory _baseURI)
        ERC1155("")
        Ownable(msg.sender)
    {
        require(_hashToken != address(0), "Invalid token");
        require(_teamWallet != address(0), "Invalid wallet");

        hashToken = IERC20(_hashToken);
        teamWallet = _teamWallet;
        baseURI = _baseURI;
        _setDefaultRoyalty(_teamWallet, ROYALTY_BPS);
    }

    // =====================================================
    // CORE
    // =====================================================

    function createMascot(uint256 _price) external {
        require(hashToken.transferFrom(msg.sender, teamWallet, CREATION_FEE), "Creation fee failed");
        emit CreationFeePaid(msg.sender, CREATION_FEE);

        uint256 tokenId = nextTokenId++;

        keys[tokenId] = MascotKey({
            creator: msg.sender,
            price: _price,
            currentSupply: 1,
            totalSold: 0,
            exists: true
        });

        allTokenIds.push(tokenId);
        _mint(msg.sender, tokenId, 1, "");

        emit MascotCreated(tokenId, msg.sender, _price);
    }

    function mintKey(uint256 _tokenId) external nonReentrant {
        MascotKey storage k = keys[_tokenId];
        require(k.exists, "Not exist");
        require(k.currentSupply < MAX_SUPPLY, "Sold out");

        uint256 teamFee = (k.price * 5) / 100;
        uint256 creatorShare = k.price - teamFee;

        require(hashToken.transferFrom(msg.sender, teamWallet, teamFee), "Team fee");
        require(hashToken.transferFrom(msg.sender, k.creator, creatorShare), "Creator fee");

        k.currentSupply++;
        k.totalSold++;
        _mint(msg.sender, _tokenId, 1, "");

        emit MascotMinted(_tokenId, msg.sender);
    }

    function discardMascot(uint256 _tokenId) external {
        require(balanceOf(msg.sender, _tokenId) > 0, "Not owner");
        safeTransferFrom(msg.sender, teamWallet, _tokenId, 1, "");
        emit MascotDiscarded(_tokenId, msg.sender);
    }

    function setPrice(uint256 _tokenId, uint256 _newPrice) external {
        require(keys[_tokenId].exists, "Not exist");
        require(keys[_tokenId].creator == msg.sender, "Only creator");
        keys[_tokenId].price = _newPrice;
        emit PriceUpdated(_tokenId, _newPrice);
    }

    function deleteMascot(uint256 _tokenId) external onlyOwner {
        require(keys[_tokenId].exists, "Already deleted");

        keys[_tokenId].exists = false;
        _removeFromAllTokens(_tokenId);

        emit MascotDeleted(_tokenId);
    }

    // =====================================================
    // OPTIMIZED VIEWS
    // =====================================================

    function getAllMascots(uint256 offset, uint256 limit)
        external
        view
        returns (
            uint256[] memory tokenIds,
            MascotKey[] memory details,
            uint256 total
        )
    {
        total = allTokenIds.length;

        if (offset >= total) {
            return (new uint256[](0), new MascotKey[](0), total);
        }

        uint256 end = offset + limit > total ? total : offset + limit;
        uint256 len = end - offset;

        tokenIds = new uint256[](len);
        details = new MascotKey[](len);

        for (uint256 i = 0; i < len; i++) {
            tokenIds[i] = allTokenIds[offset + i];
            details[i] = keys[tokenIds[i]];
        }

        return (tokenIds, details, total);
    }

    /// @notice Returns ONLY active (non-deleted) mascots — saves client-side filtering
    /// @dev Skips deleted tokens on-chain so frontend gets clean data
    function getActiveMascots(uint256 offset, uint256 limit)
        external
        view
        returns (
            uint256[] memory tokenIds,
            MascotKey[] memory details,
            uint256 totalActive
        )
    {
        uint256 total = allTokenIds.length;

        // First pass: count active tokens
        uint256 activeCount = 0;
        for (uint256 i = 0; i < total; i++) {
            if (keys[allTokenIds[i]].exists) {
                activeCount++;
            }
        }
        totalActive = activeCount;

        if (offset >= activeCount) {
            return (new uint256[](0), new MascotKey[](0), activeCount);
        }

        // Second pass: collect the requested slice of active tokens
        uint256 end = offset + limit > activeCount ? activeCount : offset + limit;
        uint256 len = end - offset;

        tokenIds = new uint256[](len);
        details = new MascotKey[](len);

        uint256 activeIdx = 0;
        uint256 collected = 0;

        for (uint256 i = 0; i < total && collected < len; i++) {
            if (keys[allTokenIds[i]].exists) {
                if (activeIdx >= offset) {
                    uint256 localIdx = activeIdx - offset;
                    tokenIds[localIdx] = allTokenIds[i];
                    details[localIdx] = keys[allTokenIds[i]];
                    collected++;
                }
                activeIdx++;
            }
        }

        return (tokenIds, details, activeCount);
    }

    /// @notice Batch read keys for multiple tokenIds — saves N individual RPC calls
    function batchGetKeys(uint256[] calldata _tokenIds)
        external
        view
        returns (MascotKey[] memory)
    {
        uint256 len = _tokenIds.length;
        MascotKey[] memory result = new MascotKey[](len);
        for (uint256 i = 0; i < len; i++) {
            result[i] = keys[_tokenIds[i]];
        }
        return result;
    }

    /// @notice Batch check ownership for a user across multiple tokenIds
    /// @return owned True if user has at least 1 of each tokenId
    function batchHasBalance(address _owner, uint256[] calldata _tokenIds)
        external
        view
        returns (bool[] memory)
    {
        uint256 len = _tokenIds.length;
        bool[] memory result = new bool[](len);
        for (uint256 i = 0; i < len; i++) {
            result[i] = balanceOf(_owner, _tokenIds[i]) > 0;
        }
        return result;
    }

    /// @notice Returns user mascots with full details AND batch balances
    /// @dev Uses ERC1155 balanceOf internally — single RPC for all data
    function getUserMascots(address _owner)
        external
        view
        returns (
            uint256[] memory tokenIds,
            uint256[] memory balances,
            MascotKey[] memory details
        )
    {
        uint256[] storage list = userTokens[_owner];
        uint256 len = list.length;

        tokenIds = new uint256[](len);
        balances = new uint256[](len);
        details = new MascotKey[](len);

        for (uint256 i = 0; i < len; i++) {
            tokenIds[i] = list[i];
            balances[i] = balanceOf(_owner, list[i]);
            details[i] = keys[list[i]];
        }

        return (tokenIds, balances, details);
    }

    /// @notice One-call: get user's owned mascots + their active keys + balances
    /// @dev Frontend no longer needs separate batchGetKeys or batchHasBalance calls
    function getUserMascotsCompact(address _owner)
        external
        view
        returns (
            uint256[] memory tokenIds,
            uint256[] memory balances,
            MascotKey[] memory details,
            uint256 count
        )
    {
        uint256[] storage list = userTokens[_owner];
        uint256 len = list.length;
        count = len;

        tokenIds = new uint256[](len);
        balances = new uint256[](len);
        details = new MascotKey[](len);

        for (uint256 i = 0; i < len; i++) {
            tokenIds[i] = list[i];
            balances[i] = balanceOf(_owner, list[i]);
            details[i] = keys[list[i]];
        }

        return (tokenIds, balances, details, count);
    }

    function getUserMascotCount(address _owner) external view returns (uint256) {
        return userTokens[_owner].length;
    }

    // =====================================================
    // METADATA
    // =====================================================

    function uri(uint256 _tokenId) public view override returns (string memory) {
        require(keys[_tokenId].exists, "Nonexistent token");
        return string(abi.encodePacked(baseURI, _tokenId.toString(), ".json"));
    }

    function contractURI() public view returns (string memory) {
        return collectionMetadata;
    }

    // =====================================================
    // ADMIN
    // =====================================================

    function setBaseURI(string memory _newBaseURI) external onlyOwner {
        baseURI = _newBaseURI;
    }

    function setCollectionMetadata(string memory _newMetadata) external onlyOwner {
        collectionMetadata = _newMetadata;
    }

    function setTeamWallet(address _newWallet) external onlyOwner {
        require(_newWallet != address(0), "Invalid address");
        teamWallet = _newWallet;
        _setDefaultRoyalty(_newWallet, ROYALTY_BPS);
    }

    function withdrawETH() external onlyOwner {
        (bool success, ) = payable(owner()).call{value: address(this).balance}("");
        require(success, "ETH transfer failed");
    }

    // =====================================================
    // INTERNAL
    // =====================================================

    function _removeFromAllTokens(uint256 _tokenId) private {
        uint256 len = allTokenIds.length;
        for (uint256 i = 0; i < len; i++) {
            if (allTokenIds[i] == _tokenId) {
                allTokenIds[i] = allTokenIds[len - 1];
                allTokenIds.pop();
                break;
            }
        }
    }

    function _update(
        address from,
        address to,
        uint256[] memory ids,
        uint256[] memory values
    ) internal virtual override {
        super._update(from, to, ids, values);

        if (to != address(0)) {
            for (uint256 i = 0; i < ids.length; i++) {
                if (values[i] > 0 && !userTokenExists[to][ids[i]]) {
                    userTokens[to].push(ids[i]);
                    userTokenExists[to][ids[i]] = true;
                }
            }
        }

        if (from != address(0)) {
            for (uint256 i = 0; i < ids.length; i++) {
                if (balanceOf(from, ids[i]) == 0) {
                    userTokenExists[from][ids[i]] = false;
                    _removeUserToken(from, ids[i]);
                }
            }
        }
    }

    function _removeUserToken(address user, uint256 tokenId) private {
        uint256[] storage tokens = userTokens[user];
        for (uint256 i = 0; i < tokens.length; i++) {
            if (tokens[i] == tokenId) {
                tokens[i] = tokens[tokens.length - 1];
                tokens.pop();
                break;
            }
        }
    }

    function supportsInterface(bytes4 interfaceId)
        public
        view
        override(ERC1155, ERC2981)
        returns (bool)
    {
        return super.supportsInterface(interfaceId);
    }
}
